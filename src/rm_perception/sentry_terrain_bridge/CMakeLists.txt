cmake_minimum_required(VERSION 3.8)
project(sentry_terrain_bridge)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rog_map REQUIRED)
find_package(rclcpp REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(pluginlib REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(PCL REQUIRED)
find_package(yaml_cpp_vendor REQUIRED)
find_package(OpenMP)

# 原库导出的是库名而非 CMake target，因此显式补充公共头文件需要的宏与依赖。
add_executable(terrain_node src/terrain_node.cpp)
# PCLConfig may omit the common include directory after repeated dependency lookups.
# Keep the discovered common header path explicit (no hard-coded PCL version).
target_include_directories(terrain_node PRIVATE include ${PCL_INCLUDE_DIRS} ${PCL_COMMON_INCLUDE_DIR})
target_compile_definitions(terrain_node PRIVATE USE_ROS2 ORIGIN_AT_CORNER ROOT_DIR="${CMAKE_CURRENT_SOURCE_DIR}/")
ament_target_dependencies(terrain_node rog_map rclcpp nav_msgs sensor_msgs pcl_conversions geometry_msgs std_msgs visualization_msgs)
target_link_libraries(terrain_node Eigen3::Eigen ${PCL_LIBRARIES} yaml-cpp)
if(OpenMP_CXX_FOUND)
  target_link_libraries(terrain_node OpenMP::OpenMP_CXX)
endif()

add_library(terrain_layer SHARED src/terrain_layer.cpp)
target_include_directories(terrain_layer PRIVATE include)
ament_target_dependencies(terrain_layer rclcpp nav_msgs nav2_costmap_2d pluginlib tf2 tf2_ros)
add_library(static_correction_layer SHARED src/static_correction_layer.cpp)
ament_target_dependencies(static_correction_layer rclcpp nav_msgs nav2_costmap_2d pluginlib tf2_ros)
pluginlib_export_plugin_description_file(nav2_costmap_2d terrain_plugins.xml)
install(TARGETS terrain_node DESTINATION lib/${PROJECT_NAME})
install(TARGETS terrain_layer static_correction_layer LIBRARY DESTINATION lib)
install(PROGRAMS scripts/real_passage_diagnostics.py
  DESTINATION lib/${PROJECT_NAME}
  RENAME real_passage_diagnostics)
install(DIRECTORY include/ DESTINATION include)

if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_hollow_structure test/test_hollow_structure.cpp)
  target_include_directories(test_hollow_structure PRIVATE include ${PCL_INCLUDE_DIRS} ${PCL_COMMON_INCLUDE_DIR})
  target_compile_definitions(test_hollow_structure PRIVATE USE_ROS2 ORIGIN_AT_CORNER ROOT_DIR="${CMAKE_CURRENT_SOURCE_DIR}/")
  ament_target_dependencies(test_hollow_structure rog_map rclcpp sensor_msgs)
  target_link_libraries(test_hollow_structure Eigen3::Eigen ${PCL_LIBRARIES})
  ament_add_gtest(test_clearance_validator test/test_clearance_validator.cpp)
  target_include_directories(test_clearance_validator PRIVATE include)
  ament_add_gtest(test_grid_contract test/test_grid_contract.cpp)
  target_include_directories(test_grid_contract PRIVATE include)
  ament_add_gtest(test_grid_scan test/test_grid_scan.cpp)
  target_include_directories(test_grid_scan PRIVATE include)
  ament_target_dependencies(test_grid_scan nav_msgs sensor_msgs)
endif()
ament_package()
